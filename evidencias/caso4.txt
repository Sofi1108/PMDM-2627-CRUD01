# 4. Modificar un campo conservando el identificador.
# Solo cambia NOMBRE; el ID del programa no cambia.
curl.exe -X PATCH http://localhost:3000/programas/2 \
  -H "Content-Type: application/json" \
  -d '{"nombre":"Noticias actualizadas"}'

# También se puede modificar solo la duración:
curl.exe -X PATCH http://localhost:3000/programas/2 \
  -H "Content-Type: application/json" \
  -d '{"duracionMinutos":90}'
